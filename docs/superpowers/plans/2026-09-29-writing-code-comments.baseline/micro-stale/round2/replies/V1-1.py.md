```python
"""Read rotated log files into entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    line: int
    host: str
    ts: str
    seq: int
    text: str


class ReadError(Exception):
    """The file cannot be read as a log at all."""

def read_entries(path, host):
    """The entries in the log file at path, in file order, and the count of lines skipped.

    A blank line, or one without a numeric sequence number, is skipped and counted.
    """
    try:
        with open(path, encoding="utf-8") as f:
            lines = f.readlines()
    except UnicodeDecodeError as e:
        raise ReadError(f"not text: {e}") from None
    entries = []
    skipped = 0
    # Skipped lines still advance the count, so line stays the entry's line in the file.
    for line, raw in enumerate(lines, 1):
        parts = raw.rstrip("\n").split(" ", 2)
        try:
            seq = int(parts[1])
        except (IndexError, ValueError):
            skipped += 1
            continue
        text = parts[2] if len(parts) > 2 else ""
        entries.append(Entry(path, line, host, parts[0], seq, text))
    return entries, skipped
```
