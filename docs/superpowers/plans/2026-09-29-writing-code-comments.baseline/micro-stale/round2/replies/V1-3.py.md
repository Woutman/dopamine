```python
"""Read rotated log files into entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    line: int
    host: str
    ts: str
    seq: int
    text: str


class ReadError(Exception):
    """The file cannot be read as a log at all."""

def read_entries(path, host):
    """Every entry in the log file at path, in file order, and the count of skipped lines.

    A blank line, or a line without a numeric sequence number, is not an entry:
    it is skipped and counted rather than failing the file.

    Returns:
        A tuple (entries, skipped).
    """
    try:
        with open(path, encoding="utf-8") as f:
            lines = f.readlines()
    except UnicodeDecodeError as e:
        raise ReadError(f"not text: {e}") from None
    entries = []
    skipped = 0
    for line, raw in enumerate(lines, 1):
        parts = raw.rstrip("\n").split(" ", 2)
        if len(parts) < 2 or not parts[1].isdigit():
            skipped += 1
            continue
        ts, seq = parts[0], int(parts[1])
        text = parts[2] if len(parts) == 3 else ""
        entries.append(Entry(path, line, host, ts, seq, text))
    return entries, skipped
```
