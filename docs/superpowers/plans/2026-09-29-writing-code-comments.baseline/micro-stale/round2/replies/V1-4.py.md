```python
"""Read rotated log files into entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    line: int
    host: str
    ts: str
    seq: int
    text: str


class ReadError(Exception):
    """The file cannot be read as a log at all."""


def read_entries(path, host):
    """Every entry in the log file at path, in file order, and the number of lines skipped.

    A blank line, or one without a numeric sequence number, is not an entry: it is
    skipped and counted rather than failing the file.

    Returns:
        A tuple ``(entries, skipped)``.
    """
    try:
        with open(path, encoding="utf-8") as f:
            lines = f.readlines()
    except UnicodeDecodeError as e:
        raise ReadError(f"not text: {e}") from None
    entries = []
    skipped = 0
    # line is kept so a person can find the entry in this file; the same entry sits
    # at a different line in the file's compressed copy.
    for line, raw in enumerate(lines, 1):
        parts = raw.rstrip("\n").split(" ", 2)
        if len(parts) < 2:
            skipped += 1
            continue
        try:
            seq = int(parts[1])
        except ValueError:
            skipped += 1
            continue
        text = parts[2] if len(parts) == 3 else ""
        entries.append(Entry(path, line, host, parts[0], seq, text))
    return entries, skipped
```
