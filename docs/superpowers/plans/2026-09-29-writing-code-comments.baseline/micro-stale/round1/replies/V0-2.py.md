```python
"""Read rotated log files into spool entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    # Not part of the spool key; kept so a person can find the entry in its file.
    line: int
    host: str
    ts: str
    seq: int
    text: str


def spool_key(entry):
    # A rotated file and its compressed copy are both shipped, and the same entry
    # sits at different line numbers in each. Host, timestamp and sequence number
    # match in both copies, so both map to the same key.
    return f"{entry.host}:{entry.ts}:{entry.seq}"


def read_entries(path, host):
    entries = []
    for line, raw in enumerate(open(path, encoding="utf-8"), 1):
        ts, seq, text = raw.rstrip("\n").split(" ", 2)
        entries.append(Entry(path, line, host, ts, int(seq), text))
    return entries
```
