```python
"""Read rotated log files into spool entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    line: int  # Lets a person find the entry in its file; not part of the key.
    host: str
    ts: str
    seq: int
    text: str


def spool_key(entry):
    # A rotated file and its compressed copy hold the same entry at different
    # line numbers; host, timestamp and sequence number match in both.
    return f"{entry.host}:{entry.ts}:{entry.seq}"


def read_entries(path, host):
    entries = []
    for line, raw in enumerate(open(path, encoding="utf-8"), 1):
        ts, seq, text = raw.rstrip("\n").split(" ", 2)
        entries.append(Entry(path, line, host, ts, int(seq), text))
    return entries
```
