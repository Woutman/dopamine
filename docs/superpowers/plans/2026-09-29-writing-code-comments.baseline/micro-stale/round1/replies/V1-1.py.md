```python
"""Read rotated log files into spool entries."""
from dataclasses import dataclass


@dataclass
class Entry:
    file: str
    line: int
    host: str
    ts: str
    seq: int
    text: str


def spool_key(entry):
    # A rotated file's compressed copy is shipped too, with the entry on a different line;
    # host, timestamp and sequence number are the same in both copies.
    return f"{entry.host}:{entry.ts}:{entry.seq}"


def read_entries(path, host):
    entries = []
    # The line number only helps a person find the entry in this file. It is not part of the key.
    for line, raw in enumerate(open(path, encoding="utf-8"), 1):
        ts, seq, text = raw.rstrip("\n").split(" ", 2)
        entries.append(Entry(path, line, host, ts, int(seq), text))
    return entries
```
