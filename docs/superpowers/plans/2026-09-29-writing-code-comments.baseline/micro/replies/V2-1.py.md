```python
"""When to flush a batch, when to give up on one, and where shipping resumes."""

from pathlib import Path

# Small batches make the collector's per-request cost dominate.
MAX_BATCH_BYTES = 512 * 1024
# An off-peak batch may never fill; this bounds how long it waits.
MAX_BATCH_AGE_S = 30
# Unbounded retries fill the spool; a batch past this goes to dead/ for a person.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True if ``batch`` should be sent at time ``now``.

    A batch is flushed once it holds ``MAX_BATCH_BYTES`` or has been open
    for ``MAX_BATCH_AGE_S`` seconds, whichever comes first.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and say what to do next.

    Increments ``batch.attempts``. Returns ``"dead"`` once the batch has
    failed ``MAX_ATTEMPTS`` times, otherwise ``"retry"``.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the log offset to resume from, read from the offset file at ``path``.

    Returns 0 if the file does not exist.
    """
    # Do not resume from the collector's acknowledgements: it acknowledges
    # before its own write is durable, so an ack can outlive a collector crash.
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
