```python
# shipper/batch.py
from pathlib import Path

# The collector's per-request cost dominates, so batches are flushed by size.
MAX_BATCH_BYTES = 512 * 1024
# Bounds how long an off-peak batch waits to reach MAX_BATCH_BYTES.
MAX_BATCH_AGE_S = 30
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True if the batch should be sent now.

    A batch is flushed once it holds MAX_BATCH_BYTES, or once it is
    MAX_BATCH_AGE_S seconds old, whichever comes first. `now` is in the
    same clock as `batch.created_at`.
    """
    if batch.size_bytes >= MAX_BATCH_BYTES:
        return True
    return now - batch.created_at >= MAX_BATCH_AGE_S


def after_failure(batch):
    """Record a failed send and return "retry" or "dead".

    Returns "dead" once the batch has failed MAX_ATTEMPTS times; the caller
    moves a dead batch to `dead/`, where it waits for a person.
    """
    batch.attempts += 1
    if batch.attempts >= MAX_ATTEMPTS:
        return "dead"
    return "retry"


def resume_offset(path):
    """Return the offset to resume shipping from, or 0 if `path` does not exist."""
    # Resume from this file, never from the collector's acknowledgements: the
    # collector acknowledges before its write is durable, so an acknowledged
    # batch can be lost in a collector crash. This file is written only after
    # a 200 and an fsync.
    offset_file = Path(path)
    if not offset_file.exists():
        return 0
    return int(offset_file.read_text().strip())
```
