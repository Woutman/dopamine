```python
# shipper/batch.py
from pathlib import Path

# Size is the primary trigger: small requests make the collector's per-request cost dominate.
MAX_BATCH_BYTES = 512 * 1024
# Bounds how long an off-peak batch that never fills can wait.
MAX_BATCH_AGE_S = 30
# A batch still failing after this many attempts is moved to dead/ for a person to inspect.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True when ``batch`` should be sent.

    A batch is flushed once it holds at least MAX_BATCH_BYTES, or once
    MAX_BATCH_AGE_S seconds have passed since ``batch.opened_at``.
    ``now`` is in the same clock as ``batch.opened_at``.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and say what to do next.

    Increments ``batch.attempts``. Returns ``"retry"`` while fewer than
    MAX_ATTEMPTS attempts have been made, and ``"dead"`` after that; the
    caller moves a dead batch to ``dead/``.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the log offset to resume shipping from, as read from ``path``.

    Returns 0 if the offset file does not exist.
    """
    # Do not resume from the collector's acknowledgements instead. The collector
    # acknowledges before its own write is durable, so an ack can outlive a
    # collector crash. This file is written only after a 200 and an fsync.
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
