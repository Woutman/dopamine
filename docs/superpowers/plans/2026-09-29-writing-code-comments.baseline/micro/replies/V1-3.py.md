```python
# shipper/batch.py
from pathlib import Path

# Small batches cost the collector more per request than they carry.
MAX_BATCH_BYTES = 512 * 1024
# Bounds how long an off-peak batch waits to fill.
MAX_BATCH_AGE_S = 30
# A batch that has failed this many times does not later succeed.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True if ``batch`` should be sent now.

    A batch is flushed once it holds at least MAX_BATCH_BYTES, or once
    MAX_BATCH_AGE_S seconds have passed since ``batch.opened_at``.
    ``now`` is in the same clock as ``batch.opened_at``.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and say what to do next.

    Increments ``batch.attempts``. Returns ``"dead"`` once it reaches
    MAX_ATTEMPTS, meaning the caller moves the batch to ``dead/``;
    otherwise returns ``"retry"``.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the offset to resume shipping from, read from the file at ``path``.

    Returns 0 if the file does not exist.
    """
    # Do not resume from the collector's acknowledgements instead: it
    # acknowledges before its own write is durable. This file is written
    # only after a 200 and an fsync.
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
