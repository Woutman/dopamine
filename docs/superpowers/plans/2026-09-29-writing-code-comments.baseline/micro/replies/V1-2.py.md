```python
# shipper/batch.py
from pathlib import Path

# The collector's cost is per request, so small batches are the expensive ones.
MAX_BATCH_BYTES = 512 * 1024

# Off-peak a batch may never reach MAX_BATCH_BYTES; this bounds its wait.
MAX_BATCH_AGE_S = 30

# A batch that has failed this many times does not later succeed.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True if ``batch`` should be sent now.

    ``now`` is in the same clock as ``batch.created_at``, in seconds.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.created_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and return ``"retry"`` or ``"dead"``.

    A ``"dead"`` batch belongs in ``dead/`` for a person to inspect.
    """
    batch.attempts += 1
    if batch.attempts >= MAX_ATTEMPTS:
        return "dead"
    return "retry"


def resume_offset(path):
    """Return the offset stored in the offset file at ``path``, or 0 if it does not exist."""
    # Do not resume from the collector's acknowledgements: it acknowledges
    # before its own write is durable, so an ack can outlive a collector crash.
    # This file is written only after a 200 and an fsync.
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
