```python
# shipper/batch.py
from pathlib import Path

# Small batches are dominated by the collector's per-request cost.
MAX_BATCH_BYTES = 512 * 1024
# Bounds how long an off-peak batch waits to fill.
MAX_BATCH_AGE_S = 30
# A batch that has failed this many times does not later succeed.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True if ``batch`` should be sent now.

    A batch is flushed once it holds ``MAX_BATCH_BYTES`` or once
    ``MAX_BATCH_AGE_S`` seconds have passed since it was opened.

    Args:
        batch: The open batch; uses ``batch.size_bytes`` and ``batch.opened_at``.
        now: Current time, in seconds, on the same clock as ``batch.opened_at``.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and say what to do next.

    Returns:
        ``"retry"`` to send it again, or ``"dead"`` once it has failed
        ``MAX_ATTEMPTS`` times; the caller moves a dead batch to ``dead/``
        for a person to inspect.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the log offset to resume shipping from.

    Reads the offset stored at ``path``, or returns 0 if the file does not exist.
    """
    # Do not resume from the collector's acknowledgements: it acknowledges
    # before its write is durable. This file is written only after a 200
    # and an fsync.
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
