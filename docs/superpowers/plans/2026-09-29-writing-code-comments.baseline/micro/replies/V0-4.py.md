```python
# shipper/batch.py
from pathlib import Path

# Flushing by size, not age: small batches let the collector's per-request
# cost dominate. At peak a 512 KB batch fills in about 2 s, which keeps p99
# delivery under 9 s.
MAX_BATCH_BYTES = 512 * 1024

# Off-peak a batch may never reach MAX_BATCH_BYTES; this bounds its wait.
MAX_BATCH_AGE_S = 30

# No batch that failed five times has ever gone on to succeed, and retrying
# forever lets the spool grow without limit.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True once the batch is full or has waited MAX_BATCH_AGE_S."""
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send and return "retry" or "dead".

    A "dead" batch is moved to dead/ for a person to look at; it is not
    retried.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the offset to resume shipping from, or 0 if none is recorded.

    The offset comes from this file and not from the collector's
    acknowledgements. The collector acknowledges before its own write is
    durable, so an acknowledgement can survive a collector crash that lost
    the data. The file is written only after a 200 and an fsync.
    """
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
