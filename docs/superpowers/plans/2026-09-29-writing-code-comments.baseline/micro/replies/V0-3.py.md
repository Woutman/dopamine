```python
# shipper/batch.py
from pathlib import Path

MAX_BATCH_BYTES = 512 * 1024
# Without an age ceiling, an off-peak batch might never reach MAX_BATCH_BYTES.
MAX_BATCH_AGE_S = 30
# A batch still failing after this many attempts is moved to dead/ so a person can look at it.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True when the batch is full, or has been open for MAX_BATCH_AGE_S seconds."""
    if batch.size_bytes >= MAX_BATCH_BYTES:
        return True
    return now - batch.opened_at >= MAX_BATCH_AGE_S


def after_failure(batch):
    """Count a failed send against the batch. Return "retry", or "dead" once it has used MAX_ATTEMPTS."""
    batch.attempts += 1
    if batch.attempts >= MAX_ATTEMPTS:
        return "dead"
    return "retry"


def resume_offset(path):
    """Return the offset recorded in the offset file, or 0 if the file does not exist yet.

    Resume from this file and not from the collector's acknowledgements. The
    collector acknowledges before its own write is durable, so an acknowledged
    batch can be lost when the collector crashes. The offset file is written
    only after a 200 and an fsync.
    """
    try:
        text = Path(path).read_text().strip()
    except FileNotFoundError:
        return 0
    return int(text) if text else 0
```
