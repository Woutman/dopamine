```python
# shipper/batch.py
import os

MAX_BATCH_BYTES = 512 * 1024
# Off-peak a batch may never reach MAX_BATCH_BYTES; this bounds how long it waits.
MAX_BATCH_AGE_S = 30
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True when `batch` should be sent.

    `batch` has `size_bytes` and `opened_at` (seconds, same clock as `now`).
    """
    return (
        batch["size_bytes"] >= MAX_BATCH_BYTES
        or now - batch["opened_at"] >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send on `batch` and return "retry" or "dead".

    "dead" means the caller moves the batch to `dead/` for a person to inspect.
    """
    batch["attempts"] += 1
    return "dead" if batch["attempts"] >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the byte offset to resume shipping from, or 0 if `path` does not exist."""
    # Do not resume from the collector's acknowledgements: it acknowledges before
    # its own write is durable. This file is written only after a 200 and an fsync.
    if not os.path.exists(path):
        return 0
    with open(path) as f:
        return int(f.read().strip() or 0)
```
