```python
# shipper/batch.py
import os

# Flush by size: the collector's per-request cost dominates small batches.
MAX_BATCH_BYTES = 512 * 1024
# Bounds how long an off-peak batch waits to fill.
MAX_BATCH_AGE_S = 30
# A batch that has failed this many times does not later succeed.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True when `batch` should be sent: it holds at least
    MAX_BATCH_BYTES, or it is at least MAX_BATCH_AGE_S seconds old at `now`."""
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.created_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of `batch` and say what to do next.

    Returns "retry", or "dead" once MAX_ATTEMPTS sends have failed. The
    caller moves a "dead" batch to `dead/`, where a person handles it.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the offset to resume shipping from, read from the offset file
    at `path`, or 0 if the file does not exist."""
    # Resume from this file, never from collector acknowledgements: the
    # collector acknowledges before its write is durable, so an ack can
    # outlive a collector crash. This file is written only after a 200 and
    # an fsync.
    if not os.path.exists(path):
        return 0
    with open(path) as f:
        return int(f.read().strip())
```
