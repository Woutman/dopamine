```python
from dataclasses import dataclass
from pathlib import Path

# Small batches cost the collector more per request than they save in latency.
MAX_BATCH_BYTES = 512 * 1024
# Limits how long an off-peak batch can wait to fill.
MAX_BATCH_AGE_S = 30
# A batch that has failed this many times does not succeed later.
MAX_ATTEMPTS = 5


@dataclass
class Batch:
    opened_at: float
    size_bytes: int = 0
    attempts: int = 0


def should_flush(batch: Batch, now: float) -> bool:
    """Return True when the batch is full or has been open too long.

    Args:
        batch: The batch being filled.
        now: The current time, on the same clock as ``batch.opened_at``.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch: Batch) -> str:
    """Record a failed send and decide what happens to the batch.

    Increments ``batch.attempts``.

    Returns:
        ``"retry"`` to send the batch again, or ``"dead"`` once it has failed
        ``MAX_ATTEMPTS`` times; the caller then moves it to ``dead/`` for a
        person to inspect.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path: Path) -> int:
    """Return the byte offset to resume shipping from, or 0 if none is recorded.

    Args:
        path: The offset file. It is written only after a 200 and an fsync.
    """
    # Do not resume from the collector's acknowledgements. The collector
    # acknowledges before its own write is durable, so an acknowledged
    # batch can be lost when the collector crashes.
    try:
        return int(path.read_text().strip())
    except FileNotFoundError:
        return 0
```
