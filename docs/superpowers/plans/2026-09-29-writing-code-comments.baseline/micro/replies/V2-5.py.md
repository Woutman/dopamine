```python
# shipper/batch.py
import os

# Each request costs the collector the same whatever its size, so batches flush by size.
MAX_BATCH_BYTES = 512 * 1024
# Stops a batch waiting forever off-peak, when it may never reach MAX_BATCH_BYTES.
MAX_BATCH_AGE_S = 30
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True when ``batch`` should be sent.

    A batch is sent once it holds ``MAX_BATCH_BYTES`` or has been open for
    ``MAX_BATCH_AGE_S`` seconds. ``now`` is on the same clock as
    ``batch.opened_at``.
    """
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send of ``batch`` and say what to do next.

    Returns ``"retry"`` while the batch has fewer than ``MAX_ATTEMPTS``
    attempts. After that it returns ``"dead"``, and the caller moves the
    batch to ``dead/`` for a person to look at.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the log offset to resume shipping from, or 0 if ``path`` does not exist.

    ``path`` is the offset file. It is written only after the collector
    returns 200 and the file has been fsynced.
    """
    # Do not resume from the collector's acknowledgements. The collector
    # acknowledges before its own write is durable, so a crash there can lose
    # data it has already acknowledged.
    if not os.path.exists(path):
        return 0
    with open(path) as f:
        return int(f.read().strip())
```
