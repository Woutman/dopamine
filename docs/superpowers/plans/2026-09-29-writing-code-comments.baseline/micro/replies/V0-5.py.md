# shipper/batch.py
from dataclasses import dataclass
from pathlib import Path

# Flush by size: the collector charges per request, so small batches are
# the expensive case. At peak a batch of this size fills in about 2 s.
MAX_BATCH_BYTES = 512 * 1024

# Off-peak, a batch might never reach MAX_BATCH_BYTES. This limit caps
# how long it can wait.
MAX_BATCH_AGE_S = 30.0

# In 30 days of collector logs, no batch that failed five times went on
# to succeed. Retrying without a limit filled the spool during an outage.
MAX_ATTEMPTS = 5


@dataclass
class Batch:
    opened_at: float
    size_bytes: int = 0
    attempts: int = 0


def should_flush(batch: Batch, now: float) -> bool:
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch: Batch) -> str:
    """Count the failed attempt. Return "retry", or "dead" when the caller
    should move the batch to dead/ for a person to look at."""
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path: Path) -> int:
    """Return the offset to resume shipping from. With no offset file, start at 0.

    Use this file, not the collector's acknowledgements. The collector
    acknowledges before its own write is durable, so a batch it acknowledged
    can be lost if it crashes. This file is written only after a 200 and an
    fsync.
    """
    try:
        return int(path.read_text().strip())
    except FileNotFoundError:
        return 0
