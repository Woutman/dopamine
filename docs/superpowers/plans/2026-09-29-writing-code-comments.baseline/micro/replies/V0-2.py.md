```python
"""When to flush a batch, when to give up on one, and where to resume after a restart."""

from pathlib import Path

# Small batches make the collector's per-request cost dominate. At peak a
# batch this size fills in about 2 s, which keeps p99 delivery under 9 s.
MAX_BATCH_BYTES = 512 * 1024

# Off-peak a batch may never reach MAX_BATCH_BYTES; this bounds its wait.
MAX_BATCH_AGE_S = 30

# In 30 days of collector logs, no batch that failed this many times went on
# to succeed. More retries would only grow the spool.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True once the batch is full, or has waited MAX_BATCH_AGE_S."""
    return (
        batch.size_bytes >= MAX_BATCH_BYTES
        or now - batch.opened_at >= MAX_BATCH_AGE_S
    )


def after_failure(batch):
    """Record a failed send. Return "retry", or "dead" once MAX_ATTEMPTS is reached.

    A "dead" batch goes to dead/ for a person to look at. It is not dropped.
    """
    batch.attempts += 1
    return "dead" if batch.attempts >= MAX_ATTEMPTS else "retry"


def resume_offset(path):
    """Return the offset saved at `path`, or 0 if nothing has been shipped yet.

    Resume from this file rather than from the collector's acknowledgements.
    The collector acknowledges before its own write is durable, so an
    acknowledged batch can be lost in a collector crash. This file is written
    only after a 200 and an fsync.
    """
    try:
        return int(Path(path).read_text().strip())
    except FileNotFoundError:
        return 0
```
