```python
# shipper/batch.py
from pathlib import Path

# Batches are flushed by size rather than age. Small batches let the
# collector's per-request cost dominate, and at peak a batch this size
# fills in about 2 s.
MAX_BATCH_BYTES = 512 * 1024

# Upper limit on how long an off-peak batch waits to fill.
MAX_BATCH_AGE_S = 30

# A batch that has failed this many times has not been seen to succeed
# later. Retrying it only grows the spool.
MAX_ATTEMPTS = 5


def should_flush(batch, now):
    """Return True once the batch is full or has waited MAX_BATCH_AGE_S."""
    if batch.size_bytes >= MAX_BATCH_BYTES:
        return True
    return now - batch.opened_at >= MAX_BATCH_AGE_S


def after_failure(batch):
    """Record a failed send. Return "retry", or "dead" once the batch has
    failed MAX_ATTEMPTS times. The caller moves a dead batch to dead/ for
    a person to look at."""
    batch.attempts += 1
    if batch.attempts >= MAX_ATTEMPTS:
        return "dead"
    return "retry"


def resume_offset(path):
    """Return the offset to resume from, or 0 if no offset file exists.

    Resume from this file, not from the collector's acknowledgements. The
    collector acknowledges before its own write is durable, so an
    acknowledged batch can be lost when the collector crashes. This file
    is written only after a 200 and an fsync.
    """
    p = Path(path)
    if not p.exists():
        return 0
    return int(p.read_text().strip())
```
